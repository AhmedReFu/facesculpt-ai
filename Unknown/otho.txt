// FaceVisionScreen.tsx

import React, { useEffect, useRef, useState } from "react";
import { View, Text, StyleSheet } from "react-native";
import { Frame, useCameraDevice } from "react-native-vision-camera";
import {
  Camera as FaceCamera,
  Face,
} from "react-native-vision-camera-face-detector";

const MIN_FACE_RATIO = 0.06; // face jodi etar cheye choto hoy -> too far
const MAX_FACE_RATIO = 0.35; // face jodi etar cheye boro hoy -> too close

type FaceStatus = "NO_FACE" | "TOO_FAR" | "TOO_CLOSE" | "OK" | "NO_LANDMARKS";

const FaceFindDetect: React.FC = () => {
  const device = useCameraDevice("front");
  const cameraRef = useRef<any>(null);

  const [faces, setFaces] = useState<Face[]>([]);
  const [faceStatus, setFaceStatus] = useState<FaceStatus>("NO_FACE");

  const faceDetectionOptions = useRef<any>({
    performanceMode: "fast",
    landmarkMode: "all", // jeno landmarks ase
    contourMode: "none",
    classificationMode: "none",
    minFaceSize: 0.2,
    trackingEnabled: true,
    // latest version hole ekhane cameraFacing diye dile valo:
    // cameraFacing: "front",
  }).current;

  useEffect(() => {
    (async () => {
      const status: any = await (FaceCamera as any).requestCameraPermission();
      console.log("permission:", status);
    })();
  }, [device]);

  const hasRequiredLandmarks = (face: Face) => {
    const lm = face.landmarks;
    if (!lm) return false;

    return (
      lm.LEFT_EYE &&
      lm.RIGHT_EYE &&
      lm.NOSE_BASE &&
      lm.MOUTH_LEFT &&
      lm.MOUTH_RIGHT
    );
  };

  const evaluateFaceDistance = (face: Face, frame: Frame): FaceStatus => {
    const { width, height } = face.bounds;
    const faceArea = width * height;
    const frameArea = frame.width * frame.height;

    const ratio = faceArea / frameArea;

    if (ratio < MIN_FACE_RATIO) return "TOO_FAR";
    if (ratio > MAX_FACE_RATIO) return "TOO_CLOSE";
    return "OK";
  };

  const handleFacesDetection = (facesDetected: Face[], frame: Frame) => {
    setFaces(facesDetected);

    if (!facesDetected || facesDetected.length === 0) {
      setFaceStatus("NO_FACE");
      return;

    // dhore nilam first face e kaj korbo
    const face = facesDetected[0];

    if (!hasRequiredLandmarks(face)) {
      setFaceStatus("NO_LANDMARKS");
      return;
    }

    const distanceStatus = evaluateFaceDistance(face, frame);
    setFaceStatus(distanceStatus);
  };

  if (!device) {
    return (
      <View style={styles.center}>
        <Text style={styles.text}>Loading camera...</Text>
      </View>
    );

        style={StyleSheet.absoluteFill}
        device={device}
        isActive={true}
        faceDetectionCallback={handleFacesDetection}
        faceDetectionOptions={faceDetectionOptions}
      />

        <Text style={styles.text}>Faces detected: {faces.length}</Text>
        <Text style={styles.textSmall}>{statusTextMap[faceStatus]}</Text>
      </View>
    </View>
  );
};

  const [faceStatus, setFaceStatus] = useState<FaceStatus>("NO_FACE");

  const faceDetectionOptions = useRef<any>({
    performanceMode: "fast",
    landmarkMode: "all", // landmark gula lagbe
    contourMode: "none",
    classificationMode: "all", // jodi probabilities use korte chao
    minFaceSize: 0.2,
    trackingEnabled: true,
  }).current;

  useEffect(() => {
    (async () => {
      const status: any = await (FaceCamera as any).requestCameraPermission();
      console.log("permission:", status);
    })();
  }, [device]);

  // 1) full face clear kina check
  const isFaceClear = (face: Face) => {
    const lm: any = face.landmarks;
    if (!lm) return false;

    // basic: 2 eye + nose + mouth corners
    const hasEyes = lm.LEFT_EYE && lm.RIGHT_EYE;
    const hasNose = lm.NOSE_BASE;
    const hasMouth =

      face.leftEyeOpenProbability == null ||
      face.leftEyeOpenProbability > 0.3;

      face.rightEyeOpenProbability == null ||
      face.rightEyeOpenProbability > 0.3;

    // ba smilingProbability use-less hoye jabe
    const smileOk =
      face.smilingProbability == null || face.smilingProbability >= 0.0;

      return;

      return;

      return;

    TOO_CLOSE: "Ektu dure giye thako",
    FACE_NOT_CLEAR: "Full face, eyes, nose, lips clear kore dekhao",

import { View, Text, StyleSheet } from "react-native";

  Face,

  | "TOO_FAR"
  | "TOO_CLOSE"
  | "FACE_NOT_CLEAR"

  const device = useCameraDevice("front");
  const cameraRef = useRef<any>(null);

  const [faces, setFaces] = useState<Face[]>([]);
  const [faceStatus, setFaceStatus] = useState<FaceStatus>("NO_FACE");

    performanceMode: "fast",
    landmarkMode: "all",      // sob landmark chai
    contourMode: "none",
    classificationMode: "all", // probabilities (eye, smile) pabo
    minFaceSize: 0.2,
    trackingEnabled: true,
  }).current;

  useEffect(() => {
    (async () => {
      const status: any = await (FaceCamera as any).requestCameraPermission();
      console.log("permission:", status);
    })();
  }, [device]);

  // helper: probability jodi null/undefined hoy -> FAIL
  const probOk = (value: number | null | undefined, min = 0.5) => {
    if (value == null) return false;
    return value >= min;
  };

  // 1) full face clear kina check (eye + nose + lip + ear + prob)
  const isFaceClear = (face: Face) => {
    const lm: any = face.landmarks;
    if (!lm) return false;

    const hasEyes = lm.LEFT_EYE && lm.RIGHT_EYE;
    const hasNose = lm.NOSE_BASE;
    const hasMouth = lm.MOUTH_LEFT && lm.MOUTH_RIGHT;
    const hasEars = lm.LEFT_EAR && lm.RIGHT_EAR;

    if (!hasEyes || !hasNose || !hasMouth || !hasEars) {
      // jodi kono ekta part-er landmark na ashe -> clear na
      return false;
    }

    // eye open probability jodi properly na ashe /kom hoy -> fail
    if (!probOk(face.leftEyeOpenProbability, 0.5)) return false;
    if (!probOk(face.rightEyeOpenProbability, 0.5)) return false;

    // smilingProbability often mouth detect-er sathe link thake,
    // etakeo ekto threshold diye check kori
    if (!probOk(face.smilingProbability, 0.2)) return false;

    return true;

  // 2) distance check

    const { width, height } = face.bounds;
    const faceArea = width * height;

    const ratio = faceArea / frameArea;

    // console.log("face ratio:", ratio.toFixed(2));

    if (ratio < MIN_FACE_RATIO) return "TOO_FAR";
    if (ratio > MAX_FACE_RATIO) return "TOO_CLOSE";
    return "OK";

    setFaces(facesDetected);

    // 0) kono face nai
    if (!facesDetected || facesDetected.length === 0) {
      setFaceStatus("NO_FACE");
      return;
    }

    // 1) ekdom 1 ta face chara onno face thakle fail
    if (facesDetected.length !== 1) {
      setFaceStatus("MULTIPLE_FACES");
      return;
    }

    const face = facesDetected[0];

    // 2) face clear kina (eye + nose + lip + ear + prob)
    if (!isFaceClear(face)) {
      setFaceStatus("FACE_NOT_CLEAR");
      return;
    }

    // 3) distance check
    const distanceStatus = evaluateFaceDistance(face, frame);
    setFaceStatus(distanceStatus);
  };

  if (!device) {
    return (
      <View style={styles.center}>
        <Text style={styles.text}>Loading camera...</Text>
      </View>
    );
  }

    NO_FACE: "Face dekha jacche na",
    MULTIPLE_FACES: "Shudhu tomar ekta face rekho (onno ke dure rakho)",
    TOO_FAR: "Ar ektu kache asho",
    TOO_CLOSE: "Ektu dure giye thako",

    OK: "Perfect! Face scan ready ✅",

      <FaceCamera
        ref={cameraRef}
        style={StyleSheet.absoluteFill}
        device={device}
        isActive={true}
        faceDetectionCallback={handleFacesDetection}
        faceDetectionOptions={faceDetectionOptions}
      />

      const status: any = await (FaceCamera as any).requestCameraPermission();
      console.log("permission:", status);
    })();

  const probOk = (value: number | null | undefined, min = 0.5) => {
    if (value == null) return false;
    return value >= min;
  };

  // 1) full face clear kina check (eye + nose + lip + ear + prob)
  const isFaceClear = (face: Face) => {
    const lm: any = face.landmarks;
    if (!lm) return false;

    const hasEyes = lm.LEFT_EYE && lm.RIGHT_EYE;
    const hasNose = lm.NOSE_BASE;
    const hasMouth = lm.MOUTH_LEFT && lm.MOUTH_RIGHT;
    const hasEars = lm.LEFT_EAR && lm.RIGHT_EAR;

    if (!hasEyes || !hasNose || !hasMouth || !hasEars) {
      // jodi kono ekta part-er landmark na ashe -> clear na
      return false;
    }

    // eye open probability jodi properly na ashe /kom hoy -> fail
    if (!probOk(face.leftEyeOpenProbability, 0.02)) return false;
    if (!probOk(face.rightEyeOpenProbability, 0.02)) return false;

    // smilingProbability often mouth detect-er sathe link thake,
    // etakeo ekto threshold diye check kori
    if (probOk(face.smilingProbability, 0.4)) return false;

    return true;
  };

  // 2) distance check
  const evaluateFaceDistance = (face: Face, frame: Frame): FaceStatus => {
    const { width, height } = face.bounds;
    const faceArea = width * height;
    const frameArea = frame.width * frame.height;
    const ratio = faceArea / frameArea;

    // console.log("face ratio:", ratio.toFixed(2));

    if (ratio < MIN_FACE_RATIO) return "TOO_FAR";
    if (ratio > MAX_FACE_RATIO) return "TOO_CLOSE";
    return "OK";
  };

  const handleFacesDetection = (facesDetected: Face[], frame: Frame) => {
    setFaces(facesDetected);

    // 0) kono face nai
    if (!facesDetected || facesDetected.length === 0) {
      setFaceStatus("NO_FACE");
      return;

    // 1) ekdom 1 ta face chara onno face thakle fail
    if (facesDetected.length !== 1) {

      return;

    // 2) face clear kina (eye + nose + lip + ear + prob)
    if (!isFaceClear(face)) {
      setFaceStatus("FACE_NOT_CLEAR");
      return;
    }

    // 3) distance check
    const distanceStatus = evaluateFaceDistance(face, frame);
    setFaceStatus(distanceStatus);
  };

  if (!device) {
    return (
      <View style={styles.center}>
        <Text style={styles.text}>Loading camera...</Text>
      </View>
    );
  }

  const statusTextMap: Record<FaceStatus, string> = {
    NO_FACE: "Face dekha jacche na",
    MULTIPLE_FACES: "Shudhu tomar ekta face rekho (onno ke dure rakho)",
    TOO_FAR: "Ar ektu kache asho",
    TOO_CLOSE: "Ektu dure giye thako",
    FACE_NOT_CLEAR:
      "Full face clear rakho: chokh, naak, lips, duita ear visible, light valo rakho",
    OK: "Perfect! Face scan ready ✅",
  };

  return (
    <View style={styles.container}>
      <FaceCamera
        ref={cameraRef}
        style={StyleSheet.absoluteFill}
        device={device}
        isActive={true}
        faceDetectionCallback={handleFacesDetection}
        faceDetectionOptions={faceDetectionOptions}
      />

      <View style={styles.overlay}>
        <Text style={styles.text}>Faces detected: {faces.length}</Text>
        <Text style={styles.textSmall}>{statusTextMap[faceStatus]}</Text>
      </View>
    </View>
  );
};
